# Data Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A `/data` page that shows every metric with records as a widget, and `/data/:key` as the widget's expanded raw mode (sources, conflicts, coverage, lazy-loaded raw rows, CSV/JSON export).

**Architecture:** Two read actions (`get-data-catalog`, `get-health-raw`) return plain rows scoped by owner; the page reuses `HealthWidget` for charts and renders the rows with toolkit tables. Export is built in the browser from the loaded windows. `WidgetMenu` gains a "Show data" link so dashboards open the same expanded view.

**Tech Stack:** `@agent-native/core` actions + `useActionQuery`, Drizzle (PGlite in tests), `@vitls/health-ingest` catalog, toolkit `Table`/`Card`/`DropdownMenu`, React Router v8, vitest.

**Spec:** `docs/superpowers/specs/2026-10-10-data-page-design.md`

All paths are relative to `apps/app`; run commands from `apps/app`. Commit only the files each task names.

## Global Constraints

- Every query filters by `owner_email` (`eq(t.ownerEmail, owner.userEmail)`); `pnpm agent-native:doctor` (`no-unscoped-queries`) must stay clean.
- `get-health-raw` windows are at most 366 days; the page asks for 92.
- Winner per day/time uses `rank()` from `server/lib/health/read.ts` (export it; do not copy the priority list).
- i18n: the `data` block goes into all 13 `app/i18n/*.ts` in the same task; the catalog test stays green.
- No new abstractions beyond `shared/data-catalog.ts` (groups + default widget), `DataCatalog`, `MetricDetail`, `lib/export.ts`.

## Review Focus

1. A metric key that is not in the catalog (`/data/nonsense`): the action returns 400 and the page shows "not found", no crash. (Test in Task 2, manual in Task 7.)
2. Two sources reporting the same day with equal priority rank (two `manual`-ranked providers): the first encountered wins deterministically (rows are ordered by `source_id`) and exactly one row per day is `winner`. (Test in Task 2.)
3. `before` earlier than the first record: `rows: []`, `nextBefore: null`, the sentinel stops. (Test in Task 2.)
4. Export of a metric with thousands of rows: the walk loads every window before the download; the button shows progress and cannot be double-clicked. (Manual in Task 7 with the dev data: `intake_energy` has 73 rows, fine; reason about larger sets.)
5. A user with no sources at all: the catalog page renders the empty state ("No data yet"), not a blank page. (Test in Task 1, manual with a fresh user if available.)

```ts
import { METRICS } from "@vitls/health-ingest";
import { count, eq, max, min } from "drizzle-orm";

import { getDb, schema } from "../../db";

export interface DataCatalog {
  sources: { id: string; provider: string; via: string; label: string | null; lastSyncAt: string | null; status: string; rows: number }[];
  metrics: { key: string; kind: "daily" | "point" | "intraday"; unit: string; rows: number; firstDay: string | null; lastDay: string | null }[];
  sections: Record<"sleep" | "workouts" | "meals", { rows: number; firstDay: string | null; lastDay: string | null }>;
}

const day = (value: Date | string | null | undefined) => (value ? new Date(value).toISOString().slice(0, 10) : null);

/** Counts behind the Data page: one round trip per table, all scoped by owner. */
export async function getDataCatalog(userEmail: string): Promise<DataCatalog> {
  const db = getDb();
  const s = schema.healthSources;
  const sources = await db
    .select({ id: s.id, provider: s.provider, via: s.via, label: s.label, lastSyncAt: s.lastSyncAt, status: s.status })
    .from(s)
    .where(eq(s.ownerEmail, userEmail));

  const d = schema.dailyMetrics;
  const daily = await db
    .select({ metric: d.metric, sourceId: d.sourceId, rows: count(), first: min(d.day), last: max(d.day) })
    .from(d)
    .where(eq(d.ownerEmail, userEmail))
    .groupBy(d.metric, d.sourceId);
  const p = schema.pointMetrics;
  const points = await db
    .select({ metric: p.metric, sourceId: p.sourceId, rows: count(), first: min(p.at), last: max(p.at) })
    .from(p)
    .where(eq(p.ownerEmail, userEmail))
    .groupBy(p.metric, p.sourceId);
  const i = schema.intradaySeries;
  const intraday = await db
    .select({ metric: i.metric, sourceId: i.sourceId, rows: count(), first: min(i.day), last: max(i.day) })
    .from(i)
    .where(eq(i.ownerEmail, userEmail))
    .groupBy(i.metric, i.sourceId);

  const section = async (t: typeof schema.sleepSessions | typeof schema.workouts) => {
    const [row] = await db
      .select({ rows: count(), first: min(t.startAt), last: max(t.endAt) })
      .from(t)
      .where(eq(t.ownerEmail, userEmail));
    const perSource = await db.select({ sourceId: t.sourceId, rows: count() }).from(t).where(eq(t.ownerEmail, userEmail)).groupBy(t.sourceId);
    return { summary: { rows: Number(row?.rows ?? 0), firstDay: day(row?.first), lastDay: day(row?.last) }, perSource };
  };
  const sleep = await section(schema.sleepSessions);
  const workouts = await section(schema.workouts);
  const m = schema.meals;
  const [mealRow] = await db
    .select({ rows: count(), first: min(m.localDate), last: max(m.localDate) })
    .from(m)
    .where(eq(m.ownerEmail, userEmail));

  const rowsBySource = new Map<string, number>();
  for (const r of [...daily, ...points, ...intraday, ...sleep.perSource, ...workouts.perSource]) {
    rowsBySource.set(r.sourceId, (rowsBySource.get(r.sourceId) ?? 0) + Number(r.rows));
  }
  const byMetric = new Map<string, { rows: number; first: string | null; last: string | null }>();
  for (const r of [...daily, ...points, ...intraday]) {
    const cur = byMetric.get(r.metric) ?? { rows: 0, first: null, last: null };
    const first = day(r.first), last = day(r.last);
    byMetric.set(r.metric, {
      rows: cur.rows + Number(r.rows),
      first: cur.first && first ? (cur.first < first ? cur.first : first) : (cur.first ?? first),
      last: cur.last && last ? (cur.last > last ? cur.last : last) : (cur.last ?? last),
    });
  }

  return {
    sources: sources.map((src) => ({ ...src, rows: rowsBySource.get(src.id) ?? 0 })),
    metrics: METRICS.map((def) => {
      const agg = byMetric.get(def.key);
      return { key: def.key, kind: def.kind, unit: def.unit, rows: agg?.rows ?? 0, firstDay: agg?.first ?? null, lastDay: agg?.last ?? null };
    }),
    sections: {
      sleep: sleep.summary,
      workouts: workouts.summary,
      meals: { rows: Number(mealRow?.rows ?? 0), firstDay: day(mealRow?.first), lastDay: day(mealRow?.last) },
    },
  };
}
```

```ts
import { fail } from "@agent-native/core/action";
import { getMetric } from "@vitls/health-ingest";
import { and, asc, desc, eq, gte, lt, min } from "drizzle-orm";

import type { MealView } from "../../../shared/meals";
import { getDb, schema } from "../../db";
import type { OwnerScope } from "../ingest/devices";
import { listMeals } from "../meals/meals";
import { rank } from "./read";

export const RAW_SECTIONS = ["sleep", "workouts", "meals"] as const;
export type RawSection = (typeof RAW_SECTIONS)[number];
export interface RawQuery { metric?: string; section?: RawSection; before?: string; days?: number }
export interface RawSource { id: string; provider: string; via: string; label: string | null }
export type RawRow =
  | { kind: "daily"; day: string; value: number; unit: string; source: RawSource; winner: boolean; updatedAt: string }
  | { kind: "point"; at: string; value: number; unit: string; source: RawSource; winner: boolean; updatedAt: string }
  | { kind: "sleep"; id: string; start: string; end: string; durationS: number; stages: unknown; scores: unknown; source: RawSource; updatedAt: string }
  | { kind: "workout"; id: string; start: string; end: string; type: string; distanceM: number | null; energyKcal: number | null; avgHr: number | null; maxHr: number | null; zones: unknown; sets: unknown; source: RawSource; updatedAt: string }
  | { kind: "meal"; meal: MealView };
export interface RawResult { window: { from: string; to: string }; rows: RawRow[]; sources: RawSource[]; nextBefore: string | null }

const DAY_MS = 86_400_000;
const MAX_DAYS = 366;
const isoDay = (ms: number) => new Date(ms).toISOString().slice(0, 10);
const iso = (value: Date | string) => new Date(value).toISOString();

/** Unresolved rows for one metric or section, newest first, in a window of `days` ending before `before` (exclusive). */
export async function getHealthRaw(owner: OwnerScope, query: RawQuery): Promise<RawResult> {
  if ((query.metric ? 1 : 0) + (query.section ? 1 : 0) !== 1) fail("Pass exactly one of `metric` or `section`", { statusCode: 400, errorCode: "invalid_query" });
  const def = query.metric ? getMetric(query.metric) : undefined;
  if (query.metric && !def) fail(`Unknown metric: ${query.metric}`, { statusCode: 400, errorCode: "unknown_metric" });
  if (def?.kind === "intraday") fail("Intraday series have no raw view yet", { statusCode: 400, errorCode: "unsupported_metric" });
  const days = Math.min(Math.max(1, Math.floor(query.days ?? 92)), MAX_DAYS);
  const beforeMs = query.before ? Date.parse(`${query.before}T00:00:00Z`) : Date.parse(`${isoDay(Date.now())}T00:00:00Z`) + DAY_MS;
  if (!Number.isFinite(beforeMs)) fail("`before` must be YYYY-MM-DD", { statusCode: 400, errorCode: "invalid_query" });
  const fromMs = beforeMs - days * DAY_MS;
  const window = { from: isoDay(fromMs), to: isoDay(beforeMs - DAY_MS) };

  const db = getDb();
  const s = schema.healthSources;
  const sourceRows = await db.select({ id: s.id, provider: s.provider, via: s.via, label: s.label }).from(s).where(eq(s.ownerEmail, owner.userEmail));
  const sourceOf = new Map(sourceRows.map((src) => [src.id, src]));
  const source = (id: string): RawSource => sourceOf.get(id) ?? { id, provider: "unknown", via: "unknown", label: null };
  const used = new Set<string>();
  const rows: RawRow[] = [];
  let nextBefore: string | null = null;

  if (def?.kind === "daily") {
    const t = schema.dailyMetrics;
    const found = await db
      .select({ day: t.day, value: t.value, unit: t.unit, sourceId: t.sourceId, updatedAt: t.updatedAt })
      .from(t)
      .where(and(eq(t.ownerEmail, owner.userEmail), eq(t.metric, def.key), gte(t.day, window.from), lt(t.day, isoDay(beforeMs))))
      .orderBy(desc(t.day), asc(t.sourceId));
    const winnerByDay = new Map<string, string>();
    for (const r of found) {
      const cur = winnerByDay.get(String(r.day));
      if (!cur || rank(source(r.sourceId).provider) < rank(source(cur).provider)) winnerByDay.set(String(r.day), r.sourceId);
    }
    for (const r of found) {
      used.add(r.sourceId);
      rows.push({ kind: "daily", day: String(r.day), value: r.value, unit: r.unit, source: source(r.sourceId), winner: winnerByDay.get(String(r.day)) === r.sourceId, updatedAt: r.updatedAt });
    }
    const [older] = await db.select({ first: min(t.day) }).from(t).where(and(eq(t.ownerEmail, owner.userEmail), eq(t.metric, def.key), lt(t.day, window.from)));
    nextBefore = older?.first ? window.from : null;
  } else if (def?.kind === "point") {
    const t = schema.pointMetrics;
    const found = await db
      .select({ at: t.at, value: t.value, unit: t.unit, sourceId: t.sourceId, updatedAt: t.updatedAt })
      .from(t)
      .where(and(eq(t.ownerEmail, owner.userEmail), eq(t.metric, def.key), gte(t.at, new Date(fromMs)), lt(t.at, new Date(beforeMs))))
      .orderBy(desc(t.at), asc(t.sourceId));
    const winnerAt = new Map<string, string>();
    for (const r of found) {
      const key = iso(r.at);
      const cur = winnerAt.get(key);
      if (!cur || rank(source(r.sourceId).provider) < rank(source(cur).provider)) winnerAt.set(key, r.sourceId);
    }
    for (const r of found) {
      used.add(r.sourceId);
      rows.push({ kind: "point", at: iso(r.at), value: r.value, unit: r.unit, source: source(r.sourceId), winner: winnerAt.get(iso(r.at)) === r.sourceId, updatedAt: r.updatedAt });
    }
    const [older] = await db.select({ first: min(t.at) }).from(t).where(and(eq(t.ownerEmail, owner.userEmail), eq(t.metric, def.key), lt(t.at, new Date(fromMs))));
    nextBefore = older?.first ? window.from : null;
  } else if (query.section === "sleep") {
    const t = schema.sleepSessions;
    const found = await db.select().from(t).where(and(eq(t.ownerEmail, owner.userEmail), gte(t.startAt, new Date(fromMs)), lt(t.startAt, new Date(beforeMs)))).orderBy(desc(t.startAt));
    for (const r of found) {
      used.add(r.sourceId);
```

```ts
/** Browser-side downloads: the Data page assembles CSV/JSON from the rows it already holds; no server endpoint. */
export function downloadFile(name: string, mime: string, body: string): void {
  const url = URL.createObjectURL(new Blob([body], { type: mime }));
  const a = document.createElement("a");
  a.href = url;
  a.download = name;
  a.click();
  URL.revokeObjectURL(url);
}

const cell = (value: unknown): string => {
  const text = value === null || value === undefined ? "" : typeof value === "object" ? JSON.stringify(value) : String(value);
  return /[",\n]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text;
};

/** Flat rows → CSV with a header from the union of keys, in first-seen order. */
export function toCsv(rows: Record<string, unknown>[]): string {
  const keys: string[] = [];
  for (const row of rows) for (const key of Object.keys(row)) if (!keys.includes(key)) keys.push(key);
  return [keys.join(","), ...rows.map((row) => keys.map((key) => cell(row[key])).join(","))].join("\n");
}
```

```ts
import { callAction } from "@agent-native/core/client/hooks";

import { downloadFile } from "@/lib/export";

import type { DataCatalog } from "../../../server/lib/health/catalog";
import type { RawResult } from "../../../server/lib/health/raw";

/** Walks every window of one key (metric or section) until nextBefore is null; newest first. */
export async function loadAllRows(key: string): Promise<RawResult["rows"]> {
  const query = key === "sleep" || key === "workouts" || key === "meals" ? { section: key } : { metric: key };
  const rows: RawResult["rows"] = [];
  let before: string | undefined;
  for (let guard = 0; guard < 400; guard++) {
    const page = (await callAction("get-health-raw", { ...query, before, days: 366 } as never, { method: "GET" })) as RawResult;
    rows.push(...page.rows);
    if (!page.nextBefore) break;
    before = page.nextBefore;
  }
  return rows;
}

/** One JSON file with every metric and section that has rows. */
export async function exportEverything(catalog: DataCatalog, t: (key: string) => string): Promise<void> {
  const keys = [
    ...catalog.metrics.filter((m) => m.rows > 0 && m.kind !== "intraday").map((m) => m.key),
    ...(["sleep", "workouts", "meals"] as const).filter((s) => catalog.sections[s].rows > 0),
  ];
  const out: Record<string, unknown> = { exportedAt: new Date().toISOString(), sources: catalog.sources };
  for (const key of keys) out[key] = await loadAllRows(key);
  downloadFile(`vitls-data-${new Date().toISOString().slice(0, 10)}.json`, "application/json", JSON.stringify(out, null, 2));
  void t;
}
```

```tsx
function useRawWindows(dataKey: string) {
  const [pages, setPages] = useState<RawResult[]>([]);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);
  const query = isSection(dataKey) ? { section: dataKey } : { metric: dataKey };
  const last = pages[pages.length - 1];
  const done = pages.length > 0 && last!.nextBefore === null;
  const loadMore = async () => {
    if (loading || done) return;
    setLoading(true);
    try {
      const page = (await callAction("get-health-raw", { ...query, before: last?.nextBefore ?? undefined, days: 92 } as never, { method: "GET" })) as RawResult;
      setPages((prev) => [...prev, page]);
    } catch (failure) {
      setError((failure as Error & { actionMessage?: string }).actionMessage ?? (failure as Error).message);
    } finally {
      setLoading(false);
    }
  };
  useEffect(() => {
    setPages([]);
    setError(null);
  }, [dataKey]);
  useEffect(() => {
    if (pages.length === 0 && !loading && !error) void loadMore();
  }, [pages.length, loading, error]); // eslint-disable-line react-hooks/exhaustive-deps
  const rows = useMemo(() => pages.flatMap((p) => p.rows), [pages]);
  const sources = useMemo(() => {
    const map = new Map<string, RawResult["sources"][number]>();
    for (const p of pages) for (const s of p.sources) map.set(s.id, s);
    return [...map.values()];
  }, [pages]);
  return { rows, sources, loading, done, error, loadMore };
}

      if (entries.some((e) => e.isIntersecting)) void loadMore();

function rowWhen(r: RawRow): string {
  switch (r.kind) {
    case "daily": return r.day;
    case "point": return r.at.slice(0, 16).replace("T", " ");
    case "meal": return `${r.meal.localDate} ${r.meal.mealType ?? ""}`.trim();
    default: return `${r.start.slice(0, 16).replace("T", " ")} → ${r.end.slice(11, 16)}`;
  }
}

function rowValue(r: RawRow, key: string): string {
  switch (r.kind) {
    case "daily":
    case "point": { const p = formatParts(key, r.value); return `${p.value} ${p.unit}`.trim() + ` (${r.value} ${r.unit})`; }
    case "sleep": return `${(r.durationS / 3600).toFixed(1)} h · ${Array.isArray(r.stages) ? r.stages.length : 0} stages`;
    case "workout": return [r.type, r.distanceM ? `${(r.distanceM / 1000).toFixed(1)} km` : null, r.energyKcal ? `${Math.round(r.energyKcal)} kcal` : null, r.avgHr ? `${Math.round(r.avgHr)} bpm` : null].filter(Boolean).join(" · ");
    case "meal": return `${r.meal.items.map((i) => i.name).join(", ")} · ${Math.round(r.meal.totals.energyKcal ?? 0)} kcal`;
  }
}

/** One flat object per row for CSV. */
function flatten(r: RawRow): Record<string, unknown> {
  switch (r.kind) {
    case "daily": return { kind: r.kind, day: r.day, value: r.value, unit: r.unit, provider: r.source.provider, via: r.source.via, winner: r.winner, updatedAt: r.updatedAt };
    case "point": return { kind: r.kind, at: r.at, value: r.value, unit: r.unit, provider: r.source.provider, via: r.source.via, winner: r.winner, updatedAt: r.updatedAt };
    case "sleep": return { kind: r.kind, start: r.start, end: r.end, durationS: r.durationS, stages: r.stages, scores: r.scores, provider: r.source.provider, via: r.source.via, updatedAt: r.updatedAt };
    case "workout": return { kind: r.kind, start: r.start, end: r.end, type: r.type, distanceM: r.distanceM, energyKcal: r.energyKcal, avgHr: r.avgHr, maxHr: r.maxHr, zones: r.zones, sets: r.sets, provider: r.source.provider, via: r.source.via, updatedAt: r.updatedAt };
    case "meal": return { kind: r.kind, eatenAt: r.meal.eatenAt, localDate: r.meal.localDate, mealType: r.meal.mealType, source: r.meal.source, items: r.meal.items.map((i) => `${i.name} ${i.amount} ${i.unit}`).join("; "), energyKcal: r.meal.totals.energyKcal };
  }
}
```
